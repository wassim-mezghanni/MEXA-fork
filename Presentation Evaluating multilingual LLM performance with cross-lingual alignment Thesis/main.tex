% 2. Motivation: How do Multilingual LLMs Share Knowledge?
% ------------------------------------------------------------
\begin{frame}{Motivation: Multilingual Knowledge in Foundation Models}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Multilingual LLMs:}
      \begin{itemize}\small
        \item Pre-training data is over 80\% English.
        \item Models still transfer knowledge to non-English and low-resource languages.
      \end{itemize}
      \vspace{0.2em}
      \textbf{The Evaluation Problem:}
      \begin{itemize}\small
        \item Downstream benchmarks mix representation quality with generation capacity, decoding choice, and prompt sensitivity.
        \item Most benchmarks cover fewer than 100 languages.
      \end{itemize}
      \vspace{0.3em}
      \textbf{\color{TUMBlue}Core Question:} How do foundation models structure internal concept spaces across languages without relying on generation?
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Research Questions :}
      \begin{enumerate}\small
        \item \textbf{Scaling and Depth:} How alignment changes with parameter size and layer depth.
        \item \textbf{Sparse MoE Routing:} Whether conditional routing preserves a shared space.
        \item \textbf{Pivot Invariance:} Whether representations depend on an English hub.
        \item \textbf{Fertility Penalty:} The effect of subword fragmentation in non-Latin scripts.
        \item \textbf{Downstream Validity:} How well alignment predicts task reasoning.
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

% 3. Methodology: MEXA Framework
% ------------------------------------------------------------
\begin{frame}{Methodology: The MEXA Framework}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Position-Weighted Pooling}
        \centering
        \vspace{0.2em}
        {\small $\displaystyle e^l = \sum_{t=1}^T w_t h_t^l, \quad w_t = \frac{t}{\sum_{k=1}^T k}$}
        \vspace{0.2em}
      \end{block}
      \vspace{0.3em}
      \begin{itemize}\small
        \item Aggregates token vectors at each layer $l \in [0, L]$.
        \item Linear weighting reflects causal attention context accumulation.
        \item Fully training-free and prompt-invariant.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{2. Bidirectional Retrieval Metric}
        \centering
        \vspace{0.2em}
        {\footnotesize $\displaystyle \mu C(l) = \frac{1}{n} \sum_{i=1}^n \mathbf{1}\!\left(c_{ii}(l) > \max_{j \neq i} \{c_{ij}(l), c_{ji}(l)\}\right)$}
        \vspace{0.2em}
      \end{block}
      \vspace{0.3em}
      \begin{itemize}\small
        \item Checks if parallel sentences are mutual nearest neighbors ($P@1$).
        \item \textbf{Peak Score:} $\mu_{\text{Max}} = \max_l \mu C(l)$.
        \item \textbf{Mean Score:} $\mu_{\text{Mean}} = \frac{1}{L} \sum_{l=1}^L \mu C(l)$.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{takeawaybox}
    \textbf{Reference Paper:} \textit{MEXA: Multilingual Evaluation of English-Centric LLMs via Cross-Lingual Alignment} Kargaran, Modarressi, Nikeghbal, Diesner, Yvon, and Sch\"utze (\textbf{Findings of ACL 2025}).
  \end{takeawaybox}
\end{frame}

% 4. Corpora: FLORES-200 vs. Super-Parallel Bible Corpus

\begin{frame}{Experimental Scope: Two Complementary Corpora}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{FLORES-200 (Modern Benchmark):}
      \begin{itemize}\small
        \item \textbf{Coverage:} 204 language varieties across multiple scripts.
        \item \textbf{Quality:} Professional human translations from English Wikipedia articles.
        \item \textbf{Domain:} Contemporary encyclopedic text.
        \item \textbf{Evaluation Split:} $n = 100$ parallel sentences per language.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Super-Parallel Bible Corpus (sPBC):}
      \begin{itemize}\small
        \item \textbf{Coverage:} Over 1,400 languages, reaching rare and low-resource varieties.
        \item \textbf{Domain:} Historical and religious text with distinct syntax.
        \item \textbf{Role:} The only multi-way parallel resource for hundreds of low-resource languages.
        \item \textbf{Evaluation Split:} $n = 103$ parallel verses.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.3em}
  \begin{takeawaybox}
    \textbf{Design:} Combining modern prose with a 1,400-language corpus tests whether alignment holds across domain shifts and rare languages.
  \end{takeawaybox}
\end{frame}

% 6. Finding 1: Parameter Scaling & Layer Trajectories
% ------------------------------------------------------------
\begin{frame}{Finding 1: Parameter Scaling and Layer Trajectories}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{figures/fig_parameter_scaling.pdf}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Layerwise Representation Dynamics:}
      \begin{itemize}\small
        \item \textbf{Capacity Scaling:} In Qwen 3, peak alignment grows with parameter size: 0.6B ($0.35$) $\rightarrow$ 1.7B ($0.49$) $\rightarrow$ 4B ($0.44$) $\rightarrow$ 8B ($0.57$).
        \item \textbf{Model Comparison:} Qwen 3.5 9B reaches $\mu_{\text{Max}} = 0.7809$, ahead of Llama 3.1 8B ($0.6735$) and Mistral 7B ($0.4934$).
        \item \textbf{Upper-Middle Peak:} Causal decoders peak at \textbf{60\% to 80\% relative depth}, then diverge at the final layers for vocabulary projection.
        \item \textbf{Sentence Encoders:} Peak early ($<30\%$ depth, $\mu > 0.95$) because of explicit contrastive loss.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Causal decoders build a shared concept space in upper-middle layers, while dedicated sentence encoders align representations early through contrastive objectives.
  \end{takeawaybox}
\end{frame}

% 7. Finding 2: Cross-Domain Robustness
% ------------------------------------------------------------
\begin{frame}{Finding 2: Cross-Domain Robustness (FLORES vs. Bible)}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{figures/fig_flores_vs_bible_all_labeled.pdf}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Cross-Domain Stability Analysis:}
      \begin{itemize}\small
        \item \textbf{Stable Rankings:} Model rankings correlate strongly at $\boldsymbol{\rho = 0.98}$ ($p < 10^{-10}$) across 16 models on 116 FLORES and 101 Bible languages.
        \item \textbf{Domain Score Shift:} Absolute alignment is lower on biblical text (e.g., Llama 3.1 8B drops from $0.67$ to $0.42$) due to archaic vocabulary and syntax.
        \item \textbf{Apertus Performance:} Apertus models retain higher relative scores on biblical text ($\mu_{\text{Max}} \le 0.50$).
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Relative model rankings remain consistent between modern and biblical text, showing that MEXA captures underlying multilingual capacity rather than domain familiarity.
  \end{takeawaybox}
\end{frame}

% 8. Finding 3: Sparse Mixture-of-Experts Routing
% ------------------------------------------------------------
\begin{frame}{Finding 3: Sparse MoE Routing Dynamics}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{figures/fig_moe_layer_trajectories.pdf}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Router Behavior and Language Allocation:}
      \begin{itemize}\small
        \item \textbf{Sparse Efficiency:} Mixtral 8x7B ($0.5385$) and 8x22B ($0.6120$) match or exceed dense Mistral 7B ($0.4934$) using sparse computation.
        \item \textbf{Functional Gating:} Routers assign tokens based on syntactic and contextual function, not language identity.
        \item \textbf{Balanced Routing:} Intermediate layers distribute tokens evenly across all 8 experts ($\boldsymbol{\approx 12.5\%}$ each) regardless of script.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Sparse conditional execution preserves representation sharing just as effectively as dense models, without splitting languages across separate experts.
  \end{takeawaybox}
\end{frame}

% 9. Finding 4: Pivot Invariance & Representation Geometry
% ------------------------------------------------------------
\begin{frame}{Finding 4: Pivot Invariance and Concept Space Geometry}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Testing the English Hub Hypothesis:}
      \begin{itemize}\small
        \item Evaluated six pivots: English, French, German, Arabic, Chinese, and Basque (isolate).
        \item \textbf{Standard Retrieval:} Non-English pivots match or exceed English (e.g., French $\mu_{\text{Max}} = 0.8034$ vs.\ English $0.7809$ on Qwen 3.5 9B).
        \item \textbf{Mean-Centering ($\mathbf{e} - \bar{\mathbf{e}}_L$):} Subtracting per-language centroid vectors removes language-specific cluster offsets.
        \item \textbf{Convergence:} Alignment across all six pivots converges to a uniform ceiling of $\boldsymbol{\mu_{\text{Max}} \approx 0.94\text{ to }0.96}$.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.42\textheight,keepaspectratio]{figures/fig_embedding_projections.pdf}
      \vspace{-0.2em}
      {\scriptsize \textbf{Centering Effect:} Subtracting centroids $\bar{\mathbf{e}}_L$ aligns parallel sentence representations onto a single shared space.}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Internal representations are not English-centric. Intermediate layers form a shared concept space where semantic equivalence determines proximity.
  \end{takeawaybox}
\end{frame}

% 10. Finding 5: The Tokenizer Fertility Tax
% ------------------------------------------------------------
\begin{frame}{Finding 5: The Tokenizer Fertility Tax}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.45\textheight,keepaspectratio]{figures/fig_fertility_alignment.pdf}
    \end{column}
    \begin{column}{0.50\textwidth}
      \textbf{Subword Fragmentation Penalty:}
      \begin{itemize}\small
        \item \textbf{Cross-Script Trend:} Subword fertility has a strong negative log-linear relationship with alignment across 204 FLORES languages.
        \item \textbf{Within-Script Correlations:} Higher fertility correlates with lower alignment within scripts: Cyrillic ($r = -0.93$), Arabic ($r = -0.88$), and Latin ($r = -0.76$).
        \item \textbf{Mechanism:} Splitting words into short character fragments reduces token weights in positional pooling ($t/T$) and adds noise.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} High subword fertility in non-Latin scripts artificially lowers sentence alignment, independent of the model's actual reasoning capacity.
  \end{takeawaybox}
\end{frame}

% 12. Summary of Contributions
% ------------------------------------------------------------
\begin{frame}{Summary of Thesis Contributions}
  \begin{enumerate}\small
    \item \textbf{Architecture Hierarchy:} Evaluated 16 foundation models across 204 FLORES varieties and 1,400 Bible languages, measuring representation alignment for Qwen 3.5, Apertus, and Gemma 4.
    \vspace{0.2em}
    \item \textbf{MoE Routing Preservation:} Found that sparse MoE models match dense alignment without segregating languages across experts, routing tokens by syntactic function.
    \vspace{0.2em}
    \item \textbf{Language-Agnostic Concept Space:} Tested six pivot languages; mean-centering shows representations converge to a shared, multi-directional concept space ($\mu \approx 0.95$).
    \vspace{0.2em}
    \item \textbf{Fertility Tax Analysis:} Quantified how subword fragmentation lowers alignment in non-Latin scripts independently of model reasoning capacity.
    \vspace{0.2em}
    \item \textbf{Predictive Task Validity:} Confirmed that internal representation alignment directly predicts downstream zero-shot accuracy at both model and language levels.
  \end{enumerate}
\end{frame}

% 13. Limitations & Future Directions
% ------------------------------------------------------------
\begin{frame}{Limitations and Future Directions}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Limitations:}
      \begin{itemize}\small
        \item \textbf{Parallel Corpus Domain:} Biblical text contains archaic vocabulary and specialized syntax.
        \item \textbf{Pooling Confounding:} High fertility dilutes individual token weights under linear pooling.
        \item \textbf{Compute Overhead:} Extracting hidden states across all layers in 70B+ models requires substantial GPU memory.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Future Directions:}
      \begin{itemize}\small
        \item \textbf{Fertility-Normalized Pooling:} Decouple subword fragmentation from semantic similarity metrics.
        \item \textbf{MoE Router Regularization:} Use layerwise alignment trajectories to guide expert pruning.
        \item \textbf{Causal Activation Steering:} Apply steering vectors to improve representations in low-resource scripts.
        \item \textbf{Multimodal Probing:} Extend MEXA retrieval to vision-language and speech models.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
