% 2. Motivation: How do Multilingual LLMs Share Knowledge?
% ------------------------------------------------------------
\begin{frame}{Motivation: Multilingual Knowledge in Foundation Models}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \textbf{Multilingual LLMs}
      \begin{itemize}\small
        \item Pre-training data is over 80\% English.
        \item Yet models still transfer knowledge across non-English and low-resource languages.
      \end{itemize}
      \vspace{0.2em}
      \textbf{Evaluation challenges}
      \begin{itemize}\small
        \item Downstream benchmarks conflate representation quality with generation choices, decoding strategies, and prompt sensitivity.
        \item Most benchmarks cover fewer than 100 languages.
      \end{itemize}
      \vspace{0.3em}
      \textbf{\color{TUMBlue}Core question:} How do foundation models organize multilingual concept spaces at the internal representation level?
    \end{column}
    \begin{column}{0.44\textwidth}
      \uncover<2->{
        \textbf{Research questions}
        \begin{enumerate}\small
          \item \textbf{Scale and depth:} How alignment evolves across model size and layer depth.
          \item \textbf{Sparse Mixture of Experts (MoE) routing:} Whether conditional routing preserves or splits a shared space.
          \item \textbf{Pivot invariance:} Whether internal geometry depends on an English hub.
          \item \textbf{Fertility penalty:} How subword fragmentation affects non-Latin scripts.
          \item \textbf{Downstream validity:} How well internal alignment predicts task reasoning.
        \end{enumerate}
      }
    \end{column}
  \end{columns}
\end{frame}

% 3. Methodology: MEXA Framework
% ------------------------------------------------------------
\begin{frame}{Methodology: The MEXA Framework}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Position-Weighted Pooling}
        \centering
        \vspace{0.1em}
        {\small $\displaystyle e^l = \sum_{t=1}^T w_t h_t^l, \quad w_t = \frac{t}{\sum_{k=1}^T k}$}
        \vspace{0.1em}
      \end{block}
      \vspace{0.1em}
      \begin{itemize}\scriptsize \setlength{\itemsep}{1.5pt}
        \item $h_t^l \in \mathbb{R}^d$: Hidden state of token $t$ at layer $l \in [0, L]$.
        \item $w_t$: Linear position weight (weights later tokens higher due to causal attention).
        \item $e^l$: Pooled sentence representation at layer $l$.
        \item Fully training-free and prompt-invariant.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \uncover<2->{
        \begin{block}{2. Bidirectional Retrieval Metric}
          \centering
          \vspace{0.1em}
          {\footnotesize $\displaystyle \mu C(l) = \frac{1}{n} \sum_{i=1}^n \mathbf{1}\!\left(c_{ii}(l) > \max_{j \neq i} \{c_{ij}(l), c_{ji}(l)\}\right)$}
          \vspace{0.1em}
        \end{block}
        \vspace{0.1em}
        \begin{itemize}\scriptsize \setlength{\itemsep}{1pt}
          \item $n$: Number of parallel test sentences.
          \item $c_{ij}(l) = \cos(e_i^l, e_j^{\prime\,l})$: Cross-lingual cosine similarity.
          \item $\mathbf{1}(\cdot)$: Mutual nearest-neighbor match ($i \leftrightarrow i$).
          \item \textbf{Peak alignment:} $\mu_{\text{Max}} = \max_l \mu C(l)$.
          \item \textbf{Mean alignment:} $\mu_{\text{Mean}} = \frac{1}{L} \sum_{l=1}^L \mu C(l)$.
        \end{itemize}
      }
    \end{column}
  \end{columns}
  \vspace{0.2em}
  \begin{takeawaybox}
    \textbf{Reference:} Kargaran et al., \textit{MEXA: Multilingual Evaluation of English-Centric LLMs via Cross-Lingual Alignment} (\textbf{ACL 2025 Findings}).
  \end{takeawaybox}
\end{frame}

  \end{columns}
  \vspace{0.1em}

\begin{frame}{Finding 3: Sparse MoE Routing Dynamics}
  \begin{columns}[c]

      \centering
      \includegraphics[width=\linewidth,height=0.8\textheight,keepaspectratio]{figures/fig_moe_layer_trajectories.pdf}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{itemize}\small \setlength{\itemsep}{6pt}
        \item \textbf{Sparse scaling:} Despite much larger total capacity, MoE models only slightly exceed dense baselines dense Mistral 7B ($0.49$):
          \begin{itemize}\setlength{\itemsep}{2pt}
            \item[\textbullet] Mixtral 8$\times$7B (13B active / 47B total) scores $0.54$
            \item[\textbullet] Mixtral 8$\times$22B (39B active / 141B total) reaches $0.61$
          \end{itemize}
        \item \textbf{Balanced allocation:} Intermediate layers distribute tokens evenly across all 8 experts ($\boldsymbol{\approx 12.5\%}$ each), regardless of script.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Sparse conditional execution shares representations as effectively as dense models, without siloing languages into separate experts.
  \end{takeawaybox}
\end{frame}

% 9. Finding 4: Pivot Invariance & Representation Geometry
% ------------------------------------------------------------
\begin{frame}{Finding 4: Pivot Invariance and Concept Space Geometry}
  \vspace{-0.3em}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}\footnotesize \setlength{\itemsep}{2pt}
        \item \textbf{Six pivots evaluated:} English plus 5 alternatives chosen for typological and script diversity: German (same family ), French (Latin), Arabic (Right To Left ), Chinese (logographic), Basque (language isolate ).
        \item \textbf{Dataset and coverage:} \textbf{FLORES-200}  \\ ($n = 100$ parallel sentences, 116 languages).
        \item \textbf{Non-English pivots:} Match or exceed English in standard retrieval (e.g.\ French $\mu_{\text{Max}} = 0.8034$ vs.\ English $0.7809$ on Qwen 3.5 9B).
        \item \textbf{Mean-centering ($\mathbf{e} - \bar{\mathbf{e}}_L$):} Subtracting per-language centroids removes static language offsets.
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.34\textheight,keepaspectratio]{figures/fig_embedding_projections.pdf}
      \vspace{-0.3em}
      {\scriptsize \textbf{Llama 3.1 8B t-SNE:} \\ Layer 0 (scripts) $\rightarrow$ 16 (semantics) $\rightarrow$ 32 (vocab).}
      \vspace{0.2em}
      \begin{itemize}\footnotesize \setlength{\itemsep}{2pt}
        \item \textbf{Centering effect:} Centered alignment across all six pivots converges to a shared ceiling of $\boldsymbol{\mu_{\text{Max}} \approx 0.94\text{--}0.96}$.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{-0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Internal representations are not English-dependent. Intermediate layers form a shared concept space where semantic equivalence determines proximity.
  \end{takeawaybox}
\end{frame}

% 10. Finding 5: The Tokenizer Fertility Tax
% ------------------------------------------------------------
\begin{frame}{Finding 5: The Tokenizer Fertility Tax}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.65\textheight,keepaspectratio]{figures/fig_fertility_alignment.pdf}
    \end{column}
    \begin{column}{0.50\textwidth}
      \begin{itemize}\small
        \item \textbf{Log-linear trend:} Subword fertility correlates negatively with alignment across 204 FLORES languages.
        \item \textbf{Within-script penalties:} Higher fertility tracks lower alignment within scripts: Cyrillic ($r = -0.93$), Arabic ($r = -0.88$), and Latin ($r = -0.76$).
        \item \textbf{Mechanism:} Splitting words into short fragments dilutes positional weights ($t/T$) and adds attention noise.
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \begin{takeawaybox}
    \textbf{Takeaway:} Subword fragmentation in non-Latin scripts decreases measured alignment, independent of the model's representational capacity.
  \end{takeawaybox}
\end{frame}

% 12. Summary of Contributions

\begin{frame}{Summary of Thesis Contributions}
  \begin{enumerate}\small
    \item \textbf{Empirical scale:} Evaluated 27 models across 204 FLORES varieties and 1,401 Bible languages, providing alignment profiles for Qwen 3.5, Apertus, and Gemma 4.
    \vspace{0.2em}
    \item \textbf{MoE representation preservation:} Showed that sparse MoE models match dense alignment without segregating languages across experts.
    \vspace{0.2em}
    \item \textbf{Language-agnostic concept space:} Showed that mean-centering aligns representations across six diverse pivots to a shared ceiling ($\mu \approx 0.95$).
    \vspace{0.2em}
    \item \textbf{Fertility penalty:} Quantified how subword fragmentation lowers measured alignment in non-Latin scripts.
    \vspace{0.2em}
    \item \textbf{Downstream predictive validity:} Verified that internal representation alignment predicts zero-shot task performance at both model and language levels.
  \end{enumerate}
\end{frame}

% 13. Limitations & Future Directions
% ------------------------------------------------------------
\begin{frame}{Limitations and Future Directions}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Limitations}
      \begin{itemize}\small
        \item \textbf{Parallel corpus register:} Biblical text has archaic vocabulary and specialized syntax.
        \item \textbf{Pooling sensitivity:} High fertility dilutes individual token weights under linear pooling.
        \item \textbf{Compute demands:} Extracting hidden states across all layers for 70B+ models requires substantial GPU memory.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Future directions}
      \begin{itemize}\small
        \item \textbf{Fertility-normalized pooling:} Decouple subword fragmentation from semantic similarity.
        \item \textbf{MoE router pruning:} Use layerwise alignment trajectories to guide expert selection.
        \item \textbf{Activation steering:} Apply steering vectors to improve representations for low-resource scripts.
        \item \textbf{Multimodal probing:} Extend MEXA retrieval to vision-language and speech models.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
